/* the story scene: the founding science, drawn in the hero's language (neurons, vesicles). Step 1: a neuron sends exosomes
   to a target neuron and the two fire together. Step 2 (Rett): the vesicles arrive empty, synapses fade, firing falls out of
   step. Step 3 (rescue): healthy exosomes restore cargo, synapses and synchronized firing. Step 4 (delivery): a blood vessel
   appears above, and an extracellular vesicle carrying AAV capsids crosses its wall toward the target neuron (EV-AAV-CPE).
   Vesicles flow with the scroll; every label sits in its own clear space and the step labels cross-fade in place (CY). */

function storyDraw(svg){
  var L=scL(),E={v:$("#scV",svg),s:$("#scS",svg),r:$("#scR",svg),syn:$$("#scSyn path",svg),vz:$$("#scVz .vz",svg),ev:$("#scEv",svg),
    c:[$("#scC1",svg),$("#scC2",svg),$("#scC3",svg),$("#scC4",svg)],t1:$("#scT1",svg),t2:$("#scT2",svg),sync:$("#scSync",svg),off:$("#scOff",svg),tr:$("#scTr",svg),lab:$("#scLab",svg)};
  var VP=L.vp,EV=L.ev;
  function f(n){return n.toFixed(3)}
  return function(p,b){
    if(!E.s.isConnected)return;
    var s2=seg(p,.2,.3),s3=seg(p,.45,.55),s4=seg(p,.7,.8),bad=s2*(1-s3);
    E.s.style.opacity=f(b);E.r.style.opacity=f(b*(1-.45*bad));E.tr.style.opacity=f(b*(1-.35*s4));
    E.v.style.opacity=f(b*s4);
    /* vesicles flow from the neuron to its target as the page scrolls */
    E.vz.forEach(function(g,i){var q=(i/E.vz.length+p*2.4)%1,xy=bez(VP,q),a=Math.min(1,q/.1,(1-q)/.1);
      g.setAttribute("transform","translate("+xy[0].toFixed(1)+" "+xy[1].toFixed(1)+")");g.style.opacity=f(b*a*(1-.55*s4));
      g.lastChild.style.opacity=f(1-bad);g.firstChild.classList.toggle("dim",bad>.5)});
    /* synapses: present, then fading in Rett, then restored with new ones in the rescue */
    E.syn.forEach(function(s,i){var nw=s.classList.contains("nw");s.style.opacity=f(b*(nw?s3:(1-.85*s2+.85*s3)))});
    /* the vesicle with its AAV cargo crosses the vessel wall */
    var e=bez([EV[0],EV[1],EV[1],EV[2]],seg(p,.74,.98));E.ev.setAttribute("transform","translate("+e[0].toFixed(1)+" "+e[1].toFixed(1)+")");E.ev.style.opacity=f(b*s4);
    /* step labels cross-fade in place */
    var w=[1-s2,bad,s3*(1-s4),s4];E.c.forEach(function(t,i){t.style.opacity=f(b*w[i])});
    E.t1.style.opacity=f(1-bad);E.t2.style.opacity=f(bad);E.sync.style.opacity=f(1-bad);E.off.style.opacity=f(bad);
  };
}
